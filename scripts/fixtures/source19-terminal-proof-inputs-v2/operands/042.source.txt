"""Test-only actual ecCodes source bytes; no issuer, capture or model stub.

Constructor reuses regular_ll_sfc_grib1 and the actual marine parameter IDs
from test-dmi-native-provenance.py. Normal DMI creates its own native receipts.
"""
from __future__ import annotations

import datetime as dt
import hashlib
import json
from pathlib import Path
import sys

import eccodes as ec


def main(request_path: Path) -> None:
    if sys.platform != "linux" or sys.version_info[:2] != (3, 12):
        raise RuntimeError("SOURCE_TERMINAL_NATIVE_LINUX_PYTHON312_REQUIRED")
    if ec.codes_get_api_version() != "2.48.2":
        raise RuntimeError("SOURCE_TERMINAL_NATIVE_ACTUAL_ECCODES_REQUIRED")
    request = json.loads(request_path.read_text("utf-8"))
    if set(request) != {"kind", "part", "target", "output"} or request["kind"] != "SOURCE_SYNTHETIC_TERMINAL_NATIVE_DATA":
        raise ValueError("SOURCE_TERMINAL_NATIVE_REQUEST_INVALID")
    output = Path(request["output"]).resolve(strict=True)
    if output != request_path.resolve().parent / "terminal-dmi-originals" or any(output.iterdir()):
        raise ValueError("SOURCE_TERMINAL_NATIVE_NEW_OUTPUT_REQUIRED")
    target = dt.datetime.fromisoformat(request["target"].replace("Z", "+00:00"))
    if target.tzinfo != dt.timezone.utc or target.minute or target.second or target.microsecond:
        raise ValueError("SOURCE_TERMINAL_NATIVE_TARGET_INVALID")
    run = target.replace(hour=(target.hour // 6) * 6)
    lon, lat = request["part"]["waterPoint"]
    if not 7.5 <= lon <= 10.5 or not 55.5 <= lat <= 58:
        raise ValueError("SOURCE_TERMINAL_NATIVE_PART_INVALID")
    iso = lambda value: value.isoformat(timespec="seconds").replace("+00:00", "Z")
    rows, features = [], []
    # Full selectable 118-hour window plus the existing T+3 support, not a
    # fabricated minimal candidate which would fail before reaching watchdog.
    for hour in range(121):
        valid = target + dt.timedelta(hours=hour)
        name = "DKSS_NSBS_SF_" + run.strftime("%Y%m%dT%H%M%SZ") + "_" + valid.strftime("%Y%m%dT%H%M%SZ") + ".grib"
        file = output / name
        gid = ec.codes_grib_new_from_samples("regular_ll_sfc_grib1")
        try:
            for key, value in (("Ni", 3), ("Nj", 3),
                    ("latitudeOfFirstGridPointInDegrees", lat - .1),
                    ("latitudeOfLastGridPointInDegrees", lat + .1),
                    ("longitudeOfFirstGridPointInDegrees", lon - .1),
                    ("longitudeOfLastGridPointInDegrees", lon + .1),
                    ("iDirectionIncrementInDegrees", .1), ("jDirectionIncrementInDegrees", .1),
                    ("scanningMode", 64), ("resolutionAndComponentFlags", 128),
                    ("dataDate", int(run.strftime("%Y%m%d"))),
                    ("dataTime", run.hour * 100),
                    ("stepRange", str(int((valid - run).total_seconds() // 3600)))):
                ec.codes_set(gid, key, value)
            with file.open("xb") as stream:
                # U/V are the exact existing constructor's marine parameters;
                # sea-level 82 closes normal REQUIRED_TARGETS['marine'].
                for parameter, value in ((49, .25), (50, -.5), (82, .1)):
                    ec.codes_set(gid, "indicatorOfParameter", parameter)
                    ec.codes_set_values(gid, [value] * 9)
                    stream.write(ec.codes_get_message(gid))
                stream.flush()
        finally:
            ec.codes_release(gid)
        body = file.read_bytes()
        if not 1 <= len(body) < 65536:
            raise ValueError("SOURCE_TERMINAL_NATIVE_GRIB_BOUND")
        href = "https://opendataapi.dmi.dk/v1/forecastdata/download/" + name
        row = {"relativePath": "terminal-dmi-originals/" + name,
            "bytes": len(body), "sha256": hashlib.sha256(body).hexdigest(), "href": href}
        rows.append(row)
        features.append({"type": "Feature", "id": "dkss_nsbs_SF_" + name[len("DKSS_NSBS_SF_"):],
            "collection": "dkss_nsbs", "geometry": {"type": "Point", "coordinates": [lon, lat]},
            "properties": {"datetime": iso(valid), "forecast:reference_datetime": iso(run)},
            "assets": {"data": {"href": href, "type": "application/x-grib", "roles": ["data"], "file:size": len(body)}},
            "links": []})
    result = {"kind": "SOURCE_TERMINAL_SYNTHETIC_GRIB_DATA", "schemaVersion": 1,
        "part": request["part"], "target": request["target"], "modelRun": iso(run),
        "rows": rows, "features": features, "providerRequests": 0, "nativeReceiptsIssued": False}
    with (output.parent / "terminal-dmi-protocol.json").open("x", encoding="utf-8") as stream:
        json.dump(result, stream, sort_keys=True, separators=(",", ":"))


if __name__ == "__main__":
    if len(sys.argv) != 2:
        raise ValueError("SOURCE_TERMINAL_NATIVE_ARGUMENTS")
    main(Path(sys.argv[1]))
