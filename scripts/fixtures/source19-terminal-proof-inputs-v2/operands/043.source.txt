def source_native_diagnostic_projection(stderr):
    result = {'sourceDiagnostic': None, 'rawSaveDiagnostic': None, 'firstSafeCode': 'UNKNOWN'}
    catalog = SOURCE_NATIVE_OBSERVER_CATALOG
    # Parse the already safe, existing CLI projections, not arbitrary error text
    # or the later unittest stderr. A generic FAIL can never become a reason.
    for line in stderr.splitlines():
        match = re.fullmatch(r'SOURCE_RUN_DIAGNOSTIC stage=([A-Z0-9_]+) reason=([A-Z0-9_]+) node_class=([A-Z0-9_]+) rejection=([A-Z0-9_]+)', line)
        if match and result['sourceDiagnostic'] is None:
            stage, reason, node_class, rejection = match.groups()
            if (stage in catalog['stages'] and reason in catalog['reasons']
                    and node_class in catalog['nodeClasses'] and rejection in catalog['rejections']):
                result['sourceDiagnostic'] = {'stage': stage, 'reason': reason, 'nodeClass': node_class, 'rejection': rejection}
        match = re.fullmatch(r'DMI_RAW_SAVE_DIAGNOSTIC stage=([A-Z0-9_]+) reason=([A-Z0-9_]+)', line)
        if match and result['rawSaveDiagnostic'] is None:
            stage, reason = match.groups()
            if stage in catalog['rawStages'] and reason in catalog['rawReasons']:
                result['rawSaveDiagnostic'] = {'stage': stage, 'reason': reason}
    for diagnostic in (result['sourceDiagnostic'], result['rawSaveDiagnostic']):
        if diagnostic is not None and diagnostic['reason'] != 'UNKNOWN':
            result['firstSafeCode'] = diagnostic['reason']
            break
    return result


# TEST-ONLY observation, not failure/stop/deadline/SAVE authority.
# SOURCE_NATIVE_OBSERVER_CATALOG is derived from exact existing qualified sets.
def source_native_action_observation(root, source_rows, action_exit_code, streams, transport):
    result = {'kind': 'SOURCE_NATIVE_ORIGINAL_ACTION_OBSERVATION', 'phase': 'BEFORE_SECONDARY_ASSERTIONS',
        'actionExitCode': action_exit_code, 'sourceDiagnostic': None, 'rawSaveDiagnostic': None,
        'firstSafeCode': 'UNKNOWN', 'projectedFrames': [], 'unmappedFrames': 0,
        'handshakes': [], 'beforeHStages': [], 'unknownHandshakes': 0, 'logs': {}, 'observerFailure': None,
        'rawLogsRetainedPrivatelyLocally': True, 'rawLogsUploaded': False}
    try:
        catalog = SOURCE_NATIVE_OBSERVER_CATALOG
        for stream in streams:
            stream.flush()
        with transport.lock:
            calls = list(transport.calls)
            before_h = list(transport.before_h_stages)
        result['handshakes'] = [value for value in calls if value in catalog['handshakes']][-12:]
        result['unknownHandshakes'] = sum(value not in catalog['handshakes'] for value in calls)
        result['beforeHStages'] = [value for value in before_h if value in catalog['beforeHStages']][-12:]
        rows = {row['relativePath']: row for row in source_rows}
        raw_logs = {}
        for name in ('stdout', 'stderr'):
            file = root / name
            before = file.lstat()
            if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1 or before.st_size > 8 * 1024 * 1024:
                raise ValueError('SOURCE_OBSERVER_LOG_INPUT_REFUSED')
            with file.open('rb') as stream:
                opened = os.fstat(stream.fileno()); raw = stream.read(8 * 1024 * 1024 + 1); ended = os.fstat(stream.fileno())
            after = file.lstat()
            key = lambda s: (s.st_dev, s.st_ino, s.st_mode, s.st_nlink, s.st_size, s.st_mtime_ns, s.st_ctime_ns)
            if key(before) != key(opened) or key(opened) != key(ended) or key(ended) != key(after) or len(raw) != after.st_size:
                raise ValueError('SOURCE_OBSERVER_LOG_CHANGED')
            result['logs'][name] = {'bytes': len(raw), 'sha256': sha(raw)}
            raw_logs[name] = raw.decode('utf-8', errors='replace')
        result.update(source_native_diagnostic_projection(raw_logs['stderr']))
        # Only if the existing projection is absent/UNKNOWN: at most three
        # ALREADY EMITTED frames, qualified by actual normal inventory bytes.
        # No new product logger; no raw path/message or external file is exposed.
        if result['firstSafeCode'] in ('UNKNOWN', 'SOURCE_NORMAL_PHASE_FAILED'):
            for stream_name in ('stderr', 'stdout'):
                for line in raw_logs[stream_name].splitlines():
                    node = re.fullmatch(r'\s+at .*?\(?((?:file://)?/[^\r\n()]+):(\d+):(\d+)\)?', line)
                    python = re.fullmatch(r'\s+File "(/[^"\r\n]+)", line (\d+)(?:, in [^\r\n]+)?', line)
                    if not node and not python:
                        continue
                    match = node or python
                    text = match[1]
                    if text.startswith('file://'):
                        parsed = urllib.parse.urlsplit(text)
                        if parsed.netloc or parsed.query or parsed.fragment:
                            result['unmappedFrames'] += 1; continue
                        text = urllib.parse.unquote(parsed.path)
                    file = Path(text)
                    try:
                        relative = file.relative_to(root).as_posix()
                        row = rows.get(relative)
                        if row is None or '..' in file.parts or file.resolve(strict=True) != file:
                            raise ValueError('unqualified')
                        info = file.lstat()
                        if not stat.S_ISREG(info.st_mode) or info.st_nlink != 1 or info.st_size != row['bytes']:
                            raise ValueError('unqualified')
                        body = file.read_bytes()
                        if len(body) != row['bytes'] or sha(body) != row['sha256']:
                            raise ValueError('unqualified')
                        line_number = int(match[2]); column = int(match[3]) if node else None
                        if not 1 <= line_number <= len(body.splitlines()) or column is not None and not 1 <= column <= 100000:
                            raise ValueError('unqualified')
                        if len(result['projectedFrames']) < 3:
                            result['projectedFrames'].append({'module': relative, 'sha256': row['sha256'],
                                'line': line_number, 'column': column, 'stream': stream_name})
                    except (OSError, ValueError):
                        result['unmappedFrames'] += 1
        return result
    except BaseException as error:
        # Observation must never replace the original action failure/assertion.
        value = str(error)
        result['observerFailure'] = value if value in {'SOURCE_OBSERVER_LOG_INPUT_REFUSED', 'SOURCE_OBSERVER_LOG_CHANGED'} else 'UNKNOWN'
        return result

def capture_source19_before_h_failure(root, action_exited):
    # stderr is the COMPLETE pre-existing private child capture. Never truncate,
    # echo it, copy it into a public artifact, or delete the failed own fixture.
    try:
        if action_exited is not True:
            return {'capture': 'PENDING_CHILD_STOP', 'sourceAdmission': False}
        file = root / 'stderr'
        identity = lambda s: (s.st_dev, s.st_ino, s.st_mode, s.st_nlink, s.st_size, s.st_mtime_ns, s.st_ctime_ns)
        before = file.lstat()
        if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1:
            raise ValueError('SOURCE19_PRIVATE_STDERR_NOT_OWN_REGULAR_FILE')
        with file.open('rb') as stream:
            opened = os.fstat(stream.fileno())
            raw = stream.read()
            ended = os.fstat(stream.fileno())
        if identity(before) != identity(opened) or identity(ended) != identity(file.lstat()) or identity(opened) != identity(ended) or len(raw) != before.st_size:
            raise ValueError('SOURCE19_PRIVATE_STDERR_CHANGED')
        known = set()
        for relative in (ENTRY, Path('scripts/protected-private-production-runtime.mjs'), Path('scripts/lib/copernicus_component_process.py')):
            known.update(re.findall(r"['\"](SOURCE_[A-Z0-9_]+)['\"]", (ROOT / relative).read_text(encoding='utf-8')))
        result = {'capture': 'COMPLETE_STABLE', 'stderrBytes': len(raw), 'stderrSha256': sha(raw),
            'fixedFailures': classify_source19_before_h_stderr(raw, known),
            'wholePrivateStderrRetained': True, 'sourceAdmission': False}
        # A private metadata receipt, not a replacement for the retained stderr.
        with (root / 'source19-before-h-child-failure.json').open('x', encoding='utf-8') as stream:
            json.dump(result, stream, sort_keys=True)
        return result
    except BaseException:
        return {'capture': 'UNPROVEN_RETAINED', 'wholePrivateStderrRetained': True, 'sourceAdmission': False}

def source19_observer_proc(raw, io):
    if type(raw) is not bytes or type(io) is not bytes or max(len(raw), len(io)) > 16384:
        return None
    try:
        fields = raw.rsplit(b')', 1)[1].split()
        parent, user, system, start = (int(fields[index]) for index in (1, 11, 12, 19))
        if min(parent, user, system, start) < 0 or max(parent, user + system, start) >= 2 ** 63:
            return None
        counters = {}
        for line in io.splitlines():
            key, separator, value = line.partition(b':')
            if separator and key in (b'read_bytes', b'write_bytes'):
                number = int(value)
                if number < 0 or number >= 2 ** 63 or key in counters:
                    return None
                counters[key] = number
        return {'parent': parent, 'cpu': user + system, 'start': start,
            'state': {b'R': 'RUNNING', b'S': 'SLEEPING', b'D': 'IO_WAIT', b'T': 'STOPPED',
                b't': 'STOPPED', b'Z': 'ZOMBIE', b'I': 'IDLE'}.get(fields[0], 'UNKNOWN'),
            'read': counters.get(b'read_bytes'), 'write': counters.get(b'write_bytes')}
    except (ValueError, IndexError):
        return None


def source19_observer_journal(raw):
    if type(raw) is not bytes or len(raw) > 65536:
        return {'status': 'UNKNOWN'}
    result = {'events': dict.fromkeys(('child-start', 'child-complete', 'worker-complete'), 0),
        'otherRows': 0, 'malformedRows': 0, 'partialTail': bool(raw and not raw.endswith(b'\n'))}
    for line in raw.split(b'\n')[:-1]:
        try:
            row = json.loads(line)
            event = row.get('event') if type(row) is dict else None
            if event not in result['events']:
                result['otherRows'] += 1
                continue
            if (set(row) != {'schemaVersion', 'actor', 'pid', 'event'} or type(row['schemaVersion']) is not int or row['schemaVersion'] != 1
                    or type(row['pid']) is not int or row['pid'] < 1
                    or row['actor'] != ('node' if event == 'worker-complete' else 'python')):
                raise ValueError('OBSERVATION_NOT_A_JOURNAL_PROOF')
            result['events'][event] += 1
        except (ValueError, TypeError, RecursionError):
            result['malformedRows'] += 1
    return result


class Source19ObserverTelemetry:
    ROLES = ('ACTION', 'SUPERVISOR', 'WORKER', 'CP_1', 'CP_2', 'CP_3', 'CP_4')
    EVENTS = ('H70_SEEN', 'H71_SEEN', 'NATIVE_CP_SEEN', 'RESULT_NONEMPTY', 'WORKING_INSTALL_SEEN', 'FAILURE')

    def __init__(self, started):
        self.started, self.next_sample, self.cost = started, started, 0.0
        self.events, self.roles, self.union_identity = {}, {}, None

    def event(self, name):
        if name in self.EVENTS and name not in self.events:
            self.events[name] = round(max(0, time.monotonic() - self.started) * 1000, 3)

    def register(self, role, pid):
        if role in self.ROLES and type(pid) is int and pid > 0 and role not in self.roles:
            self.roles[role] = {'pid': pid, 'first': None, 'last': None, 'samples': 0, 'history': [], 'status': 'UNKNOWN', 'parentChangedSeen': False}
            self.sample(True)

    def cp(self, pid):
        if not any(row['pid'] == pid for row in self.roles.values()):
            for role in self.ROLES[3:]:
                if role not in self.roles:
                    self.register(role, pid)
                    break

    def sample(self, force=False):
        now = time.monotonic()
        if not force and now < self.next_sample:
            return
        self.next_sample = now + 1.0
        for row in self.roles.values():
            if row['status'] == 'IDENTITY_CHANGED':
                continue
            try:
                root = Path('/proc') / str(row['pid'])
                value = source19_observer_proc((root / 'stat').read_bytes(), (root / 'io').read_bytes())
                if value is None:
                    row['status'] = 'UNKNOWN'
                elif row['first'] is not None and value['start'] != row['first']['start']:
                    row['status'] = 'IDENTITY_CHANGED'
                else:
                    row['first'] = row['first'] or value
                    row['last'], row['samples'] = value, row['samples'] + 1
                    row['history'] = (row['history'] + [(now, value)])[-4:]
                    row['lastSeenMs'] = round(max(0, now - self.started) * 1000, 3)
                    row['status'] = 'PARENT_CHANGED' if value['parent'] != row['first']['parent'] else 'SAME_IDENTITY'
                    row['parentChangedSeen'] |= row['status'] == 'PARENT_CHANGED'
            except Exception:
                row['status'] = 'NOT_PRESENT_OR_UNREADABLE'
        self.cost += time.monotonic() - now

    def bind_union(self, directory):
        try:
            value = directory.lstat()
            if stat.S_ISDIR(value.st_mode) and not directory.is_symlink():
                self.union_identity = (value.st_dev, value.st_ino)
        except OSError:
            pass

    def journal(self, directory):
        try:
            if directory is None or self.union_identity is None:
                return {'status': 'NOT_OBSERVED'}
            named = directory.lstat()
            if directory.is_symlink() or (named.st_dev, named.st_ino) != self.union_identity:
                return {'status': 'ROOT_CHANGED'}
            fd = os.open(directory / 'union-birth.jsonl', os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
            try:
                before = os.fstat(fd)
                if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1:
                    return {'status': 'UNKNOWN'}
                offset = max(0, before.st_size - 65536)
                raw = os.pread(fd, min(before.st_size, 65536), offset)
                result = source19_observer_journal(raw.partition(b'\n')[2] if offset else raw)
                after = os.fstat(fd)
                result.update({'status': 'UNSEALED_SNAPSHOT', 'bytes': before.st_size, 'tailOnly': bool(offset),
                    'changedWhileRead': (before.st_size, before.st_mtime_ns) != (after.st_size, after.st_mtime_ns)})
                return result
            finally:
                os.close(fd)
        except Exception:
            return {'status': 'UNKNOWN_RETAINED'}

    def snapshot(self, directory):
        try:
            return self._snapshot(directory)
        except Exception:
            return {'status': 'UNKNOWN', 'sourceAdmission': False}

    def _snapshot(self, directory):
        self.sample(True)
        ticks = os.sysconf('SC_CLK_TCK')
        processes = {}
        for role, row in self.roles.items():
            first, last = row['first'], row['last']
            valid = first is not None and last is not None and row['status'] != 'IDENTITY_CHANGED'
            delta = lambda key: last[key] - first[key] if valid and first[key] is not None and last[key] is not None and last[key] >= first[key] else None
            cpu = delta('cpu')
            history = row['history']
            recent = history[0][1] if valid and history else None
            recent_delta = lambda key: last[key] - recent[key] if recent is not None and last[key] is not None and recent[key] is not None and last[key] >= recent[key] else None
            recent_cpu = recent_delta('cpu')
            processes[role] = {'status': row['status'], 'samples': row['samples'], 'state': last['state'] if valid else 'UNKNOWN',
                'parentChangedSeen': row['parentChangedSeen'],
                'lastSeenMs': row.get('lastSeenMs'), 'cpuMs': round(cpu * 1000 / ticks, 3) if cpu is not None and ticks > 0 else None,
                'readBytesDelta': delta('read'), 'writeBytesDelta': delta('write'),
                'recentWindowMs': round((history[-1][0] - history[0][0]) * 1000, 3) if recent is not None else None,
                'recentCpuMs': round(recent_cpu * 1000 / ticks, 3) if recent_cpu is not None and ticks > 0 else None,
                'recentReadBytesDelta': recent_delta('read'), 'recentWriteBytesDelta': recent_delta('write')}
        return {'status': 'OBSERVATION_ONLY', 'elapsedMs': round((time.monotonic() - self.started) * 1000, 3),
            'eventsMs': dict(self.events), 'processes': processes, 'journal': self.journal(directory),
            'samplingCostMs': round(self.cost * 1000, 3), 'sourceAdmission': False}

def strict_json(path, maximum=8 * 1024 * 1024):
    info = path.lstat()
    if not stat.S_ISREG(info.st_mode) or info.st_nlink != 1 or not 2 <= info.st_size <= maximum:
        raise AssertionError('FIXTURE_REGULAR_FILE_REQUIRED')
    with path.open('rb') as stream:
        raw = stream.read(maximum + 1)
    if len(raw) != info.st_size:
        raise AssertionError('FIXTURE_BYTES_CHANGED')
    return json.loads(raw)

def descendants(parent):
    """Discovery only. Every signalled process is separately retained by pidfd."""
    result, todo = [], [parent]
    while todo:
        pid = todo.pop()
        try:
            children = Path(f'/proc/{pid}/task/{pid}/children').read_text('ascii').split()
        except (FileNotFoundError, ProcessLookupError) as error:
            if error.errno not in (errno.ENOENT, errno.ESRCH):
                raise
            continue
        for text in children:
            child = int(text)
            if child in result:
                raise AssertionError('FIXTURE_PROCESS_GRAPH_CONFLICT')
            result.append(child)
            todo.append(child)
    return result

            selector = query['document_key'][0]
            if selector.startswith('in.(') and selector.endswith(')'):
                keys = [key.strip('"') for key in selector[4:-1].split(',')]
            elif selector.startswith('eq.'):
                keys = [selector[3:]]
            else:
                raise AssertionError('ADMIN_SELECTOR_INVALID')
            known = {row['document_key']: row for row in data['admin']}
            known[data['pointer']['document_key']] = data['pointer']
            known[data['checkpointDocumentKey']] = None
            if not keys or len(set(keys)) != len(keys) or any(key not in known for key in keys):
                raise AssertionError('ADMIN_KEY_REFUSED')
            with self.lock:
                if keys == [data['checkpointDocumentKey']]: self.before_h_stages.append('CHECKPOINT_PROTECTED_READ')
                if keys == [data['pointer']['document_key']]: self.before_h_stages.append('PRIVATE_POINTER_READ')
            request.reply(200, [known[key] for key in keys if known[key] is not None])
            return 'private-pointer' if keys == [data['pointer']['document_key']] else 'admin-read'
        if method == 'GET' and not query and route == '/storage/v1/bucket/' + data['bucket']['id']:
            with self.lock: self.before_h_stages.append('PRIVATE_BUCKET_READ')
            request.reply(200, data['bucket'])
            return 'private-bucket'
        if method == 'GET' and not query and route.startswith('/storage/v1/object/authenticated/' + data['bucket']['id'] + '/'):
            key = urllib.parse.unquote(route.split('/storage/v1/object/authenticated/' + data['bucket']['id'] + '/', 1)[1])
            rows = [row for row in data['objectRows'] if row['key'] == key]
            if len(rows) != 1:
                raise AssertionError('EXACT_B_OBJECT_REQUIRED')
            row = rows[0]
            raw = (self.root / row['relativePath']).read_bytes()
            if len(raw) != row['bytes'] or sha(raw) != row['sha256']:
                raise AssertionError('EXACT_B_OBJECT_CHANGED')
            with self.lock: self.before_h_stages.append('PRIVATE_B_OBJECT_DOWNLOAD')
            request.reply(200, raw=raw)
            return 'private-B-read'
        return request.fixed_failure('UNEXPECTED_ROUTE_OR_WRITE')

    def original_destinations(self, root, protocol):
        snapshot = {}
        for row in protocol['originals']:
            relative = Path(row['relativePath'])
            self.assertFalse(relative.is_absolute() or '..' in relative.parts)
            self.assertNotIn(row['relativePath'], snapshot, 'DUPLICATE_ORIGINAL_DESTINATION')
            destination = root / relative
            try:
                before = destination.lstat()
            except FileNotFoundError:
                snapshot[row['relativePath']] = None
                continue
            self.assertTrue(stat.S_ISREG(before.st_mode) and before.st_nlink == 1,
                'ORIGINAL_DESTINATION_NOT_OWN_REGULAR_FILE')
            self.assertLessEqual(before.st_size, 64 * 1024 * 1024)
            with destination.open('rb') as stream:
                opened = os.fstat(stream.fileno())
                raw = stream.read(64 * 1024 * 1024 + 1)
                ended = os.fstat(stream.fileno())
            after = destination.lstat()
            identity = lambda value: (value.st_dev, value.st_ino, value.st_mode,
                value.st_nlink, value.st_size, value.st_mtime_ns, value.st_ctime_ns)
            self.assertEqual(identity(before), identity(opened), 'DESTINATION_OPEN_CHANGED')
            self.assertEqual(identity(opened), identity(ended), 'DESTINATION_READ_CHANGED')
            self.assertEqual(identity(ended), identity(after), 'DESTINATION_PATH_CHANGED')
            self.assertEqual(len(raw), before.st_size)
            snapshot[row['relativePath']] = {'bytes': len(raw), 'sha256': sha(raw),
                'identity': identity(before)}
        return snapshot

            root = Path(temporary)
            (root / 'stdout').write_bytes(body)
            read, state = self.parser(root, before=before)
            while state()[0] < len(body):
                read()
            return state()
